// Дисциплина M1: третья независимая реализация (после sympy и ручного

// Прямые Фано в циклической маркировке: l_p = {p, p+1, p+3} mod 7.
fn line_has(p: Int, x: Int) -> Bool {
    x == p % 7 || x == (p + 1) % 7 || x == (p + 3) % 7
}

// Третья точка прямой через пару {i,j}; -1, если пары нет на прямой p.
fn third_point(i: Int, j: Int) -> Int {
    let mut k = -1;
    let mut p = 0;
    while p < 7 {
        if line_has(p, i) && line_has(p, j) {
            let mut m = 0;
            while m < 7 {
                if line_has(p, m) && m != i && m != j {
                    k = m;
                }
                m = m + 1;
            }
        }
        p = p + 1;
    }
    k
}

// ============================================================
// Часть 1: T-226 — полярный закон и точная томография (в целых
// числах: скорости 6*r целочисленны при целых gamma).
// ============================================================
fn check_t226() -> Int {
    let mut fails = 0;

    // gamma_p = p + 1  (те же значения, что и в разобранном
    // примере §3a корпусного документа)
    let mut gamma = [0; 7];
    let mut p = 0;
    while p < 7 {
        gamma[p] = p + 1;
        p = p + 1;
    }

    // G и точечные потоки T_k
    let mut big_g = 0;
    p = 0;
    while p < 7 {
        big_g = big_g + gamma[p];
        p = p + 1;
    }
    let mut t = [0; 7];
    let mut k = 0;
    while k < 7 {
        p = 0;
        while p < 7 {
            if line_has(p, k) {
                t[k] = t[k] + gamma[p];
            }
            p = p + 1;
        }
        k = k + 1;
    }

    // (a) полярный закон: 6*r_ij = G - T_{third(i,j)} для всех 21 пар,
    //     где 6*r_ij считается напрямую из определения канала.
    let mut i = 0;
    while i < 7 {
        let mut j = i + 1;
        while j < 7 {
            let mut six_r = 0;
            p = 0;
            while p < 7 {
                let mut cnt = 0;
                if line_has(p, i) { cnt = cnt + 1; }
                if line_has(p, j) { cnt = cnt + 1; }
                if cnt == 1 {
                    six_r = six_r + gamma[p];
                }
                p = p + 1;
            }
            let kk = third_point(i, j);
            if kk < 0 || six_r != big_g - t[kk] {
                print(f"FAIL polar law at pair ({i},{j})");
                fails = fails + 1;
            }
            j = j + 1;
        }
        i = i + 1;
    }

    // (б) сумма T_k = 3G  и  сумма (G - T_k) = 4G  (эквивалент
    //     G-восстановления: sum rho = (2/3) G при делении на 6)
    let mut sum_t = 0;
    k = 0;
    while k < 7 {
        sum_t = sum_t + t[k];
        k = k + 1;
    }
    if sum_t != 3 * big_g {
        print("FAIL sum T = 3G");
        fails = fails + 1;
    }

    // (в) томография в целых числах: 2*gamma_p = ((3G - сумма T по
    //     точкам прямой p) ... ) — эквивалентная целочисленная форма
    //     gamma_p = 3*(sum rho)/2 - 3*(rho-сумма по прямой):
    //     6*gamma_p = (сумма_k (G-T_k)) - 3*(сумма_{k in l_p} (G-T_k)) ...
    //     проще напрямую: из тождества T-суммы по прямой:
    //     сумма_{k in l_p} T_k = G + 2*gamma_p  =>  gamma_p восстановим.
    p = 0;
    while p < 7 {
        let mut line_t = 0;
        k = 0;
        while k < 7 {
            if line_has(p, k) {
                line_t = line_t + t[k];
            }
            k = k + 1;
        }
        if line_t - big_g != 2 * gamma[p] {
            print(f"FAIL line-sum identity at p={p}");
            fails = fails + 1;
        }
        p = p + 1;
    }

    // (г) полярное разбиение: ровно 3 пары на точку
    k = 0;
    while k < 7 {
        let mut cls = 0;
        i = 0;
        while i < 7 {
            let mut j = i + 1;
            while j < 7 {
                if third_point(i, j) == k {
                    cls = cls + 1;
                }
                j = j + 1;
            }
            i = i + 1;
        }
        if cls != 3 {
            print(f"FAIL polar class size at k={k}");
            fails = fails + 1;
        }
        k = k + 1;
    }

    fails
}

// Циклический Хэмминг [7,4] с генератором 1 + x + x^3 (biti 0..6),
// расширенный битом чётности в позиции 7. Возвращает 16 слов (8-бит Int).
fn extended_hamming(g0: Int, g1: Int, g2: Int, g3: Int) -> [Int; 16] {
    let mut words = [0; 16];
    let mut msg = 0;
    while msg < 16 {
        let mut c = [0; 7];
        let mut s = 0;
        while s < 4 {
            if (msg / pow2(s)) % 2 == 1 {
                // прибавить сдвинутый генератор
                let gbits = [g0, g1, g2, g3];
                let mut d = 0;
                while d < 4 {
                    if gbits[d] == 1 {
                        let pos = (s + d) % 7;
                        c[pos] = 1 - c[pos];
                    }
                    d = d + 1;
                }
            }
            s = s + 1;
        }
        let mut w = 0;
        let mut parity = 0;
        let mut b = 0;
        while b < 7 {
            if c[b] == 1 {
                w = w + pow2(b);
                parity = 1 - parity;
            }
            b = b + 1;
        }
        if parity == 1 {
            w = w + pow2(7);
        }
        words[msg] = w;
        msg = msg + 1;
    }
    words
}

fn pow2(n: Int) -> Int {
    let mut r = 1;
    let mut i = 0;
    while i < n {
        r = r * 2;
        i = i + 1;
    }
    r
}

fn popcount24(x: Int) -> Int {
    let mut c = 0;
    let mut v = x;
    let mut i = 0;
    while i < 24 {
        c = c + v % 2;
        v = v / 2;
        i = i + 1;
    }
    c
}

fn xor24(a: Int, b: Int) -> Int {
    let mut r = 0;
    let mut va = a;
    let mut vb = b;
    let mut bit = 1;
    let mut i = 0;
    while i < 24 {
        if (va % 2 + vb % 2) % 2 == 1 {
            r = r + bit;
        }
        va = va / 2;
        vb = vb / 2;
        bit = bit * 2;
        i = i + 1;
    }
    r
}

fn check_t228() -> Int {
    let mut fails = 0;
    let a8 = extended_hamming(1, 1, 0, 1);   // корпусной репер: 1+x+x^3
    let b8 = extended_hamming(1, 0, 1, 1);   // зеркальный: 1+x^2+x^3

    // |A ∩ B| = 2 (нуль и все-единицы)
    let mut inter = 0;
    let mut ia = 0;
    while ia < 16 {
        let mut ib = 0;
        while ib < 16 {
            if a8[ia] == b8[ib] {
                inter = inter + 1;
            }
            ib = ib + 1;
        }
        ia = ia + 1;
    }
    if inter != 2 {
        print(f"FAIL |A∩B| = {inter}, want 2");
        fails = fails + 1;
    }

    // сумма Тюрина: 4096 слов (a+x | b+x | a+b+x), веса -> энумератор
    let mut wt = [0; 25];
    ia = 0;
    while ia < 16 {
        let mut ib = 0;
        while ib < 16 {
            let mut ix = 0;
            while ix < 16 {
                let blk1 = xor24(a8[ia], b8[ix]);
                let blk2 = xor24(a8[ib], b8[ix]);
                let blk3 = xor24(xor24(a8[ia], a8[ib]), b8[ix]);
                let word = blk1 + blk2 * 256 + blk3 * 65536;
                let w = popcount24(word);
                wt[w] = wt[w] + 1;
                ix = ix + 1;
            }
            ib = ib + 1;
        }
        ia = ia + 1;
    }

    // энумератор Голея: 1 + 759 x^8 + 2576 x^12 + 759 x^16 + x^24
    if wt[0] != 1 { print("FAIL wt0"); fails = fails + 1; }
    if wt[8] != 759 { print(f"FAIL wt8 = {wt[8]}"); fails = fails + 1; }
    if wt[12] != 2576 { print(f"FAIL wt12 = {wt[12]}"); fails = fails + 1; }
    if wt[16] != 759 { print(f"FAIL wt16 = {wt[16]}"); fails = fails + 1; }
    if wt[24] != 1 { print("FAIL wt24"); fails = fails + 1; }
    let mut w = 1;
    while w < 8 {
        if wt[w] != 0 {
            print(f"FAIL nonzero weight below 8: wt[{w}]={wt[w]}");
            fails = fails + 1;
        }
        w = w + 1;
    }

    // совершенство прокола: 4096 * (1+23+253+1771) == 2^23
    let ball = 1 + 23 + 253 + 1771;
    if 4096 * ball != 8388608 {
        print("FAIL perfection arithmetic");
        fails = fails + 1;
    }

    fails
}

// φ-условие: сумма знаков по каждой прямой чётна.
// ============================================================
fn check_simplex_diagonals() -> Int {
    let mut count = 0;
    let mut s = 0;
    while s < 128 {
        let mut ok = 1;
        let mut p = 0;
        while p < 7 {
            let b0 = (s / pow2(p % 7)) % 2;
            let b1 = (s / pow2((p + 1) % 7)) % 2;
            let b3 = (s / pow2((p + 3) % 7)) % 2;
            if (b0 + b1 + b3) % 2 == 1 {
                ok = 0;
            }
            p = p + 1;
        }
        count = count + ok;
        s = s + 1;
    }
    if count != 8 {
        print(f"FAIL simplex diagonal count = {count}, want 8");
        return 1;
    }
    0
}

fn clampf(x: Float, lo: Float, hi: Float) -> Float {

fn softplus(x: Float) -> Float {
    if x > 20.0 { x } else { ln(1.0 + exp(x)) }

// ----------------------------------------------------------------------------
// Density7: 7x7 complex density matrix, split real/imaginary storage
// (the EpistemicState<k> pattern of core.math.quantum_logic; avoids the
// V-L1 Complex/Numeric gap). Invariants hold BY CONSTRUCTION: the only
// public constructors are from_cholesky, maximally_mixed, fano_seed,
// equal_population, and the invariant-preserving channels below.
// ----------------------------------------------------------------------------

type Density7 is {
    re: [Float; 49],
    im: [Float; 49],
};

implement Density7 {
    /// Gamma = L L^dag / Tr(L L^dag) from 49 real parameters:
    /// 7 softplus diagonals + 21 lower-triangle re + 21 lower-triangle im.
    public fn from_cholesky(params: &[Float; 49]) -> Density7 {
        let mut l_re = [0.0; 49];
        let mut l_im = [0.0; 49];
        for i in 0..7 {
            l_re[i * 7 + i] = softplus(params[i]);
        }
        let mut idx = 7;
        for i in 1..7 {
            for j in 0..i {
                l_re[i * 7 + j] = params[idx];
                l_im[i * 7 + j] = params[idx + 21];
                idx = idx + 1;
            }
        }
        let mut g_re = [0.0; 49];
        let mut g_im = [0.0; 49];
        for i in 0..7 {
            for j in 0..7 {
                let mut s_re = 0.0;
                let mut s_im = 0.0;
                for k in 0..7 {
                    // L[i,k] * conj(L[j,k])
                    let a_re = l_re[i * 7 + k];
                    let a_im = l_im[i * 7 + k];
                    let b_re = l_re[j * 7 + k];
                    let b_im = l_im[j * 7 + k];
                    s_re = s_re + a_re * b_re + a_im * b_im;
                    s_im = s_im + a_im * b_re - a_re * b_im;
                }
                g_re[i * 7 + j] = s_re;
                g_im[i * 7 + j] = s_im;
            }
        }
        let mut tr = 0.0;
        for i in 0..7 {
            tr = tr + g_re[i * 7 + i];
        }
        let inv = 1.0 / tr;   // tr > 0: softplus diagonals are strictly positive
        for x in 0..49 {
            g_re[x] = g_re[x] * inv;
            g_im[x] = g_im[x] * inv;
        }
        Density7 { re: g_re, im: g_im }
    }

    public fn maximally_mixed() -> Density7 {
        let mut re = [0.0; 49];
        let im = [0.0; 49];
        for i in 0..7 {
            re[i * 7 + i] = 1.0 / 7.0;
        }
        Density7 { re: re, im: im }
    }

    /// Fano-seeded diagonal state: E, O, U populated at 1/3 each (P = 1/3).
    public fn fano_seed() -> Density7 {
        let mut re = [0.0; 49];
        let im = [0.0; 49];
        re[4 * 7 + 4] = 1.0 / 3.0;
        re[5 * 7 + 5] = 1.0 / 3.0;
        re[6 * 7 + 6] = 1.0 / 3.0;
        Density7 { re: re, im: im }
    }

    /// Equal-population stratum state: diagonal 1/7, uniform real coherence c
    /// on every off-diagonal pair. PSD for c in [0, 1/7] (I/7 + c(J - I) with
    /// eigenvalues 1/7 + 6c and 1/7 - c). On this stratum Phi = 7P - 1 exactly.
    public fn equal_population(c: Float) -> Density7 {
        let mut re = [0.0; 49];
        let im = [0.0; 49];
        for i in 0..7 {
            for j in 0..7 {
                if i == j {
                    re[i * 7 + j] = 1.0 / 7.0;
                } else {
                    re[i * 7 + j] = c;
                }
            }
        }
        Density7 { re: re, im: im }
    }

    /// Fieldwise copy. Single-pointer-arg calls are the only shape that
    /// survives the L-4-family VBC defect past the GC epoch (~10^3
    /// allocations): every two-pointer-arg call (self + by-ref array)
    /// first silently aliases its result with an argument and later
    /// crashes with a corrupted argument register. See repro notes.
    public fn copy(&self) -> Density7 {
        Density7 { re: self.re, im: self.im }
    }

    public fn trace(&self) -> Float {
        let mut t = 0.0;
        for i in 0..7 {
            t = t + self.re[i * 7 + i];
        }
        t
    }

    /// P = Tr(rho^2) = sum |g_ij|^2 (Hermitian storage).
    public fn purity(&self) -> Float {
        let mut p = 0.0;
        for x in 0..49 {
            p = p + self.re[x] * self.re[x] + self.im[x] * self.im[x];
        }
        p
    }

    /// R = 1/(7P) (T-96 canon).
    public fn reflexivity(&self) -> Float {
        1.0 / (7.0 * self.purity())
    }

    /// Canonical integration (Def. 3.4, repair R-3):
    /// Phi = sum_{i!=j}|g_ij|^2 / sum_i g_ii^2. Denominator >= 1/7 by
    /// Cauchy-Schwarz on the unit trace, so the guard never fires on a
    /// valid state.
    public fn integration(&self) -> Float {
        let mut num = 0.0;
        let mut den = 0.0;
        for i in 0..7 {
            for j in 0..7 {
                let m2 = self.re[i * 7 + j] * self.re[i * 7 + j]
                       + self.im[i * 7 + j] * self.im[i * 7 + j];
                if i == j { den = den + m2; } else { num = num + m2; }
            }
        }
        num / (den + EPS)
    }

    /// L1 coherence mass — logged diagnostic ONLY (R-3 demoted this form).
    public fn integration_l1(&self) -> Float {
        let mut phi = 0.0;
        for i in 0..7 {
            for j in 0..7 {
                if i != j {
                    let m2 = self.re[i * 7 + j] * self.re[i * 7 + j]
                           + self.im[i * 7 + j] * self.im[i * 7 + j];
                    phi = phi + sqrt(m2);
                }
            }
        }
        phi
    }

    /// Coh_E = (g_EE^2 + 2 sum_{j!=E}|g_Ej|^2) / Tr(rho^2)
    /// (axiom-septicity HS-projection canon; range [1/7, 1]).
    public fn coh_e(&self) -> Float {
        let gee = self.re[E_AXIS * 7 + E_AXIS];
        let mut off = 0.0;
        for j in 0..7 {
            if j != E_AXIS {
                let m2 = self.re[E_AXIS * 7 + j] * self.re[E_AXIS * 7 + j]
                       + self.im[E_AXIS * 7 + j] * self.im[E_AXIS * 7 + j];
                off = off + m2;
            }
        }
        (gee * gee + 2.0 * off) / (self.purity() + EPS)
    }

    /// D_diff, 7D operational form (Def. 3.5): 1 + 6 Coh_E / Coh_E^max,
    /// Coh_E^max = 1.
    public fn d_diff(&self) -> Float {
        1.0 + 6.0 * self.coh_e()
    }

    /// Per-channel stress sigma_k = clamp(1 - 7 g_kk, 0, 1) (T-92).
    public fn sigma(&self) -> [Float; 7] {
        let mut s = [0.0; 7];
        for k in 0..7 {
            s[k] = clampf(1.0 - 7.0 * self.re[k * 7 + k], 0.0, 1.0);
        }
        s
    }

    public fn is_hermitian(&self) -> Bool {
        for i in 0..7 {
            for j in 0..7 {
                if (self.re[i * 7 + j] - self.re[j * 7 + i]).abs() > 1e-9 {
                    return false;
                }
                if (self.im[i * 7 + j] + self.im[j * 7 + i]).abs() > 1e-9 {
                    return false;
                }
            }
        }
        true
    }

    /// Schur (entrywise) dephasing: populations fixed, coherence (i,j)
    /// multiplied by m[i*7+j] in (0,1]. CPTP for multiplier matrices arising
    /// from commuting-projector Lindbladians (paper §6, v2.2).
    public fn dephase(&self, m: &[Float; 49]) -> Density7 {
        let mut re = self.re;
        let mut im = self.im;
        for i in 0..7 {
            for j in 0..7 {
                if i != j {
                    re[i * 7 + j] = re[i * 7 + j] * m[i * 7 + j];
                    im[i * 7 + j] = im[i * 7 + j] * m[i * 7 + j];
                }
            }
        }
        Density7 { re: re, im: im }
    }

    /// Convex mixture toward a target (the exact regeneration step with
    /// alpha = 1 - e^{-kappa v dt}; CPTP for every dt >= 0).
    public fn mix_toward(&self, target: &Density7, alpha: Float) -> Density7 {
        let a = clampf(alpha, 0.0, 1.0);
        let mut re = [0.0; 49];
        let mut im = [0.0; 49];
        for x in 0..49 {
            re[x] = (1.0 - a) * self.re[x] + a * target.re[x];
            im[x] = (1.0 - a) * self.im[x] + a * target.im[x];
        }
        Density7 { re: re, im: im }
    }
}

    [p, (p + 1) % 7, (p + 3) % 7]

fn line_contains(p: Int, q: Int) -> Bool {
    let l = fano_line(p);
    q == l[0] || q == l[1] || q == l[2]
}

/// The discrete self-observation cycle P_Fano: every coherence x 1/3 (T-110).
/// In-place, single-argument form: passing a freshly allocated multiplier
/// array by reference here trips the L-4-family VBC defect once total
/// allocations cross a GC epoch (~t 1000-3000 in this program) -- the
/// callee reads a stale register as its second pointer argument. With the
/// constant folded in and one argument, the frame shape avoids the defect.
fn fano_cycle_inplace(rho: &mut Density7) {
    for i in 0..7 {
        for j in 0..7 {
            if i != j {
                rho.re[i * 7 + j] = rho.re[i * 7 + j] * FANO_CONTRACTION;
                rho.im[i * 7 + j] = rho.im[i * 7 + j] * FANO_CONTRACTION;
            }
        }
    }
}

/// The exact continuous Fano channel e^{dt D}: coherence (i,j) decays at
/// r_ij = (1/6) sum_{p : |l_p cap {i,j}| = 1} gamma_p (uniform rates here).
/// In-place single-pointer-arg form (L-4 family workaround; see copy()).
fn fano_exponential_inplace(rho: &mut Density7, gamma: Float, dt: Float) {
    for i in 0..7 {
        for j in 0..7 {
            if i != j {
                let mut r = 0.0;
                for p in 0..7 {
                    let on_i = line_contains(p, i);
                    let on_j = line_contains(p, j);
                    if on_i != on_j { r = r + gamma / 6.0; }
                }
                let m = exp(0.0 - r * dt);
                rho.re[i * 7 + j] = rho.re[i * 7 + j] * m;
                rho.im[i * 7 + j] = rho.im[i * 7 + j] * m;
            }
        }
    }
}

        if (d - (1.0 + 6.0 * ce)).abs() > 1e-12 { fails = fails + 1; print(f"FAIL D formula at {t}"); }

        // gamma * tau = (3/2) ln 3  =>  multiplier e^{-(2/3) gamma tau} = 1/3.
        // (Run BEFORE the in-place cycle below, while rho is untouched.)

        // defect, not the law. Measures return scalars and stay reliable;

                    fails = fails + 1;
                    print(f"FAIL cycle-time identity at {t}");
                }
            }
        }

        sum_p = sum_p + p;
        sum_r = sum_r + r;
        sum_phi = sum_phi + phi;
        sum_phi1 = sum_phi1 + phi1;
        sum_ce = sum_ce + ce;
        sum_d = sum_d + d;

        print(f"M1 SELFTEST PASS: {n_states} states, all invariants and exact laws hold");
    } else {
        print(f"M1 SELFTEST FAIL: {fails} failures");
    }
}

\section{Complete Verum API reference}
\label{app:verum-api}

This appendix gives the complete public API of \SYNARC{} as seven Verum
cogs. Each cog exports a module whose public interface is listed here.
This appendix is the authoritative \emph{normative} reference for API
names, types, and contracts; the cogs themselves are materialized
through gates M1--M3 of \S\ref{subsec:impl-status}, and no claim of a
distributed source tree is made before those gates flip (ledger item
I-8). Two grounding facts matter for the reader: the quantum substrate
this API refines already ships in the Verum standard library
(\texttt{core.math.quantum\_logic}: \texttt{EpistemicState<k>} ---
density matrices with Hermiticity, PSD and unit trace,
\texttt{CptpChannel<k>}, L\"uders measurement, Uhlmann fidelity, Bures
distance, von~Neumann entropy), so \texttt{DensityMatrix7} below is a
thin refinement of \texttt{EpistemicState<7>} rather than a from-scratch
matrix layer; and the syndrome layer of Appendix~K ships as
\texttt{core.math.coding}. Opaque type exports are shown with their
carrier elided (\texttt{is \{ /* internal */ \}}).

\begin{notebox}[Intuition --- how to read this API as a developer]
The whole runtime is seven cogs, and they stack in dependency order rather than
sprawl. At the bottom sits \texttt{uhm\_core}: the density-matrix type and the
four consciousness measures, which everything else imports and which depends on
nothing in return. Above it the dynamics cogs --- the Fano dissipator, the
regeneration channel, the Strang integrator --- turn one state into the next;
above them the cognitive cogs (the Kan-complex levels, the modalities, the
self-model) and the sensorimotor and learning cogs that couple the mind to a
world. Two facts save a reader real work. First, \texttt{DensityMatrix7} is not a
new matrix library --- it is a thin refinement of the Verum standard library's
\texttt{EpistemicState<7>}, which already ships Hermiticity, positivity, unit
trace, fidelity, Bures distance and entropy, so this API adds the \emph{seven}
specialisation and the UHM contracts, not the linear algebra underneath. Second,
every signature carries its law in its type: a \texttt{requires} line is a
precondition the caller must meet, an \texttt{ensures} line a guarantee the SMT
backend has already checked --- so the contract you read is the contract the
compiler enforces, with no gap between the documentation and the obligation.
\end{notebox}

                  + perturbation.h_dissipator.frobenius_norm()

\subsection{Development handoff: primitives required by the agentic cogs}
\label{subsec:api-agentic-primitives}

The agentic cogs of Appendices~\ref{app:active-inference},
\ref{app:morphology-design}, and \ref{app:memory-credit}
(\texttt{uhm\_active\_inference}, \texttt{uhm\_hierarchy},
\texttt{uhm\_memory}) call six helper primitives, specified here so a development
team can implement them without ambiguity. Each either refines an existing
\texttt{uhm\_core} operation or restricts a loop already defined in the body;
none introduces new mathematics.

\begin{notebox}[For the implementer --- the six helper contracts]
\textbf{\texttt{apply\_perturbation(gamma, dh) -> DensityMatrix7}} forms the
counterfactual next state $\hat{s} = \Gam \oplus \mathrm{d}h$ by applying the
three-channel Jacobian of Definition~\hyperref[def:three-channel]{12.1} to $\Gam$;
\texttt{ensures result.is\_density()}. It refines the three-channel action of
Section~\ref{sec:sensorimotor}.

\textbf{\texttt{bayes\_update(gamma, e) -> DensityMatrix7}} is the CPTP posterior
update $\mathcal{U}(\Gam \mid e)$ of Definition~\hyperref[def:three-tier]{N.1} (the
recursive Bayesian filter), a refinement of the standard-library L\"uders update
on \texttt{EpistemicState<7>}; \texttt{ensures result.is\_density()} and, on
dense evidence, agreement with the batch posterior (T-Memory-Sufficiency).

\textbf{\texttt{project\_to\_density(m) -> DensityMatrix7}} is the projection
$\Pi_{\Dens}$ of Definition~\hyperref[def:goal-inference]{L.4} (Hermitise, clip negative
eigenvalues, renormalise trace) onto $\Dens(\mathbb{C}^7)$ --- the non-expansive
nearest-density map; \texttt{ensures result.is\_density()}.

\textbf{\texttt{project\_sector\_ASD(gamma) -> DensityMatrix7}} is the sector
projector $\Pi_{\{A,S,D\}}$ onto the spatial reflex sector of
Proposition~\ref{prop:spatial-derived}: restrict to the $\{A,S,D\}$
rows/columns and renormalise; \texttt{ensures result.is\_density()}.

\textbf{\texttt{deliberate(holon, env) -> DensityMatrix7}} (async) is the
slow-loop setpoint computation of Definition~\hyperref[def:hierarchy]{M.2}: one
full-$\Gam$ / simplicial planning pass, implemented as an invocation of the
\texttt{topological\_maturation} cycle; it returns the deliberative setpoint (a
density).

\textbf{\texttt{reflex\_correct(holon, sector, setpoint, dtau, env)}} (async) is
one fast-loop step of Theorem~\hyperref[thm:hierarchical]{T-Hierarchical-Sensorimotor}: a single Section-12
sensorimotor step restricted to the $\{A,S,D\}$ sector, driving it toward
\texttt{setpoint} over period \texttt{dtau}; \texttt{ensures
holon.gamma.purity() > 2/7} (fast-loop viability).
\end{notebox}

\subsection{Summary}

The complete public API of \SYNARC{} consists of $\sim 100$ public
functions and $\sim 25$ public types, spread across the eight
cogs (\texttt{uhm\_core}, \texttt{uhm\_connes},
\texttt{uhm\_lindblad}, \texttt{uhm\_riemannian},
\texttt{uhm\_peps}, \texttt{uhm\_cognitive},
\texttt{uhm\_sensorimotor}, \texttt{uhm\_runtime}). Every function
carries a \texttt{requires}/\texttt{ensures} contract, verified by
the Verum SMT backend at compile time. The API is deliberately
small: anything not in this list either decomposes into these
primitives or is internal to a specific cog. The main runtime loop,
\texttt{topological\_maturation}, takes the simplicial cognitive
complex, the concept site, the Giry oracle, the environment, and
the configuration, and runs the seven-modality walk of
Section~\ref{subsec:maturation-cycle}.
